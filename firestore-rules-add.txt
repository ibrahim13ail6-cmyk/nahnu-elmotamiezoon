// ====== أضف الجزء ده في Firebase ← Firestore Database ← Rules ======
// داخل:  match /databases/{database}/documents { ... }
// (لو عندك قاعدة users/{uid}/{document=**} بالفعل، متكررهاش)

match /users/{uid}/{document=**} {
  allow read, write: if request.auth != null && request.auth.uid == uid;
}

// سجل أكواد الطلاب: كل طالب له كود، وأي طالب مسجّل يقدر يدوّر على كود صاحبه
match /codes/{code} {
  allow read: if request.auth != null;
  allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
  allow update, delete: if request.auth != null && resource.data.uid == request.auth.uid;

  // صندوق الوارد: طلبات الصداقة والتحدّيات والنتائج
  match /inbox/{msg} {
    allow create: if request.auth != null && request.resource.data.fromUid == request.auth.uid;
    allow read, delete: if request.auth != null
      && get(/databases/$(database)/documents/codes/$(code)).data.uid == request.auth.uid;
  }
}
// بعد اللصق اضغط Publish.
